		// "dd/MM/yyyy HH:mm"      ← Google Sheets
		// "dd.MM.yyyy HH:mm"      ← Альтернативный (русские точки)
		// "yyyy-MM-ddTHH:mm"      ← datetime-local HTML5
		// "yyyy-MM-ddTHH:mm:ss"   ← ISO full
		// "yyyy-MM-dd HH:mm"      ← Запасной

		public static DateTime KazakhstanNow()
		{
			var utcNow = DateTime.UtcNow;
			return TimeZoneInfo.ConvertTime(
				utcNow, 
				TimeZoneInfo.FindSystemTimeZoneById("Asia/Almaty")
			);
		}

		// Парсить дату в разных форматах
		public static bool TryParse(string value, out DateTime result)
		{
			// Пробует все форматы, возвращает первый успешный
		}
	}
}

📍 СЦЕНАРИЙ 1: КЛИЕНТ ДЕЛАЕТ БРОНЬ
────────────────────────────────────────────────────────────────────────

1️⃣  HTTP POST /api/reservations
	└─ Body (JSON):
	   {
		 "TablesId": "1,2",
		 "CustomerName": "Иван",
		 "CustomerPhone": "+77770000000",
		 "ScheduledAt": "25/12/2024 19:30",
		 "Section": "VIP",
		 "RemindBeforeHour": true
	   }

2️⃣  ReservationsController.CreateReservation()
	└─ Получает [FromBody] ReservationInfo request
	   └─ Валидирует:
		  ├─ Не пусто? ✓
		  ├─ Правильный формат даты? 
		  │  └─ ReservationDateTime.TryParse() ✓
		  ├─ Время >= текущее + 5 минут?
		  │  └─ ReservationDateTime.KazakhstanNow() ✓
		  ├─ В рабочее время? (12:00 - 04:00) ✓
		  ├─ Стол существует? 
		  │  └─ _sheets.TryParseTableIds() ✓
		  └─ Стол не занят?
			 └─ _sheets.IsReservationTakenAsync() ✓

3️⃣  _sheets.AppendReservationAsync(request, scheduledAt)
	└─ Сохраняет в Google Sheets таблицу "Брони!A:H"
	└─ Возвращает: AppendValuesResponse

4️⃣  Получаем TableInfo для информации о типе стола
	└─ _sheets.GetTablesAsync(scheduledAt)
	└─ Фильтруем по ID из request.TablesId
	└─ Строим typeLabel = "VIP" или "Обычный"

5️⃣  _whatsApp.SendReservationNotificationsAsync()
	└─ Отправляет два сообщения:
	   ├─ Клиенту: "Ваша бронь подтверждена..."
	   └─ Админу: "Новая бронь: Иван, стол 1-2, 19:30..."

6️⃣  Возвращаем OK ответ:
	└─ {
		 "message": "Бронь создана.",
		 "whatsAppSent": true,
		 "adminWhatsAppSent": true,
		 "update": { "updates": 1 }
	   }

📊 ЦЕПОЧКА МОДЕЛЕЙ:
   JSON (клиент)
	  ↓
   ReservationInfo (парсинг + валидация)
	  ↓
   DateTime (парсинг времени)
	  ↓
   TableInfo[] (проверка доступности)
	  ↓
   Google Sheets (сохранение)
	  ↓
   WhatsApp (отправка уведомлений)


📍 СЦЕНАРИЙ 2: ФОНОВЫЙ СЕРВИС ОТПРАВЛЯЕТ НАПОМИНАНИЕ
────────────────────────────────────────────────────────────────────────

1️⃣  ReservationReminderService.ExecuteAsync()
	└─ Каждые 5 минут:
	   ├─ sheets.GetReminderCandidatesAsync()
	   │  └─ Читает Google Sheets "Брони!A:H"
	   │  └─ Фильтрует: брони через ~60 минут
	   │  └─ Возвращает ReminderCandidate[]
	   │
	   ├─ ReminderCandidate[] содержит:
	   │  ├─ SheetRowNumber = 5 (чтобы потом обновить)
	   │  ├─ CustomerPhone = "+77770000000"
	   │  ├─ CustomerName = "Иван"
	   │  ├─ ScheduledAt = "25/12/2024 19:30"
	   │  └─ ReminderSentCell = "H5" (ячейка для отметки)
	   │
	   └─ Для каждого ReminderCandidate:
		  ├─ _whatsApp.SendReminderBeforeHourAsync(reminder, ...)
		  │  └─ Отправляет: "Иван! Напоминаем, что в 19:30 у вас бронь на стол 1-2"
		  │
		  └─ sheets.MarkReminderSentAsync(sheetRowNumber)
			 └─ Обновляет в Google Sheets H{n} = "sent"
			 └─ Чтобы не отправить дважды

📊 ЦЕПОЧКА МОДЕЛЕЙ:
   Google Sheets (чтение брони)
	  ↓
   ReminderCandidate[] (парсинг + фильтрация)
	  ↓
   WhatsApp (отправка)
	  ↓
   Google Sheets (отметка как отправленное)


📍 СЦЕНАРИЙ 3: ПОЛУЧИТЬ ДОСТУПНЫЕ СТОЛЫ
────────────────────────────────────────────────────────────────────────

1️⃣  HTTP GET /api/tables?scheduledAt=25/12/2024 19:30
	└─ Query параметр: scheduledAt (опционально)

2️⃣  TablesController.GetTables()
	└─ if (scheduledAt != null)
		  ReservationDateTime.TryParse(scheduledAt) → DateTime
	   else
		  targetTime = null

3️⃣  _sheets.GetTablesAsync(scheduledAt: targetTime)
	└─ Читает "Столики!A2:C100" из Google Sheets
	└─ Для каждой строки создаёт TableInfo:
	   {
		 Id: 1,
		 Type: TableType.Обычный,
		 Seats: 4,
		 Status: "free",
		 NextReservationHours: 2.5
	   }
	└─ Если scheduledAt указано, проверяет брони на это время
	└─ Обновляет Status = "occupied" если занято

4️⃣  Возвращаем JSON ответ:
	└─ [
		 { "id": 1, "type": "Обычный", "seats": 4, "status": "free", ... },
		 { "id": 2, "type": "VIP", "seats": 2, "status": "occupied", ... }
	   ]

📊 ЦЕПОЧКА МОДЕЛЕЙ:
   HTTP параметры (scheduledAt строка)
	  ↓
   DateTime (парсинг)
	  ↓
   Google Sheets (чтение столов и брони)
	  ↓
   TableInfo[] (построение списка)
	  ↓

ReservationMapper — это сердце трансформаций:

   ВХОДЯЩИЕ МОДЕЛИ             MAPPER               ИСХОДЯЩИЕ МОДЕЛИ
   ─────────────────────────────────────────────────────────────────

   ReservationInfo     ──────> ToCreateReservationRequest ──────> CreateReservationRequest

   ReservationInfoDto  ──────> ToReminderCandidate() ──────> ReminderCandidate
					   ─────────────────────────────────  (для напоминаний)

   ReservationResult   ──────> IsOperationSuccessful() ──────> bool (успех?)

   List<ReservationInfoDto> ──> ToReminderCandidates() ──────> ReminderCandidate[]

ПОЧЕМУ МАППИРОВАНИЕ?
├─ Разные системы используют разные форматы
├─ Google Sheets ≠ POS ≠ HTTP API
├─ Маппер — адаптер между слоями
└─ Изменения в одной системе не ломают другую

public class MyPosAdapter : IPosAdapter
{
	public string ProviderName => "MyPOS";

	public async Task<ReservationResult> ReserveTableAsync(
		CreateReservationRequest request,
		CancellationToken ct = default)
	{
		try
		{
			// Ваш код: отправить в MyPOS API
			var response = await _http.PostAsJsonAsync(
				"https://mypos-api.com/reservations",
				new {
					restaurant = request.RestaurantId,
					customer = request.Customer.Name,
					phone = request.Customer.Phone,
					datetime = request.ReservationTime,
					guests = request.GuestsCount,
					table = request.TableId
				},
				ct
			);

			var result = await response.Content.ReadAsAsync<dynamic>(ct);

			return response.IsSuccessStatusCode
				? new ReservationResult(
					Success: true,
					ExternalReservationId: result.reservationId,
					ErrorMessage: null)
				: new ReservationResult(
					Success: false,
					ExternalReservationId: null,
					ErrorMessage: result.error);
		}
		catch (Exception ex)
		{
			return new ReservationResult(
				Success: false,
				ExternalReservationId: null,
				ErrorMessage: ex.Message);
		}
	}

	// ... остальные методы ...
}
